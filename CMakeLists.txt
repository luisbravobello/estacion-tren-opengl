cmake_minimum_required(VERSION 3.16)
project(EstacionTren LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenGL REQUIRED)
find_package(GLUT REQUIRED)
add_executable(estacion_tren estacion_tren.cpp)
target_link_libraries(estacion_tren PRIVATE OpenGL::GL OpenGL::GLU GLUT::GLUT)
if(WIN32)
  target_link_libraries(estacion_tren PRIVATE winmm)
endif()
if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  target_link_options(estacion_tren PRIVATE /SAFESEH:NO)
endif()
